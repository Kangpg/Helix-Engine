include(${HELIX_SOURCE_ROOT}/cmake/HelixFlatBuffers.cmake)
set(_schema ${HELIX_SOURCE_ROOT}/csharp/benchmarks/Helix.SerializationBenchmarks/world_state.fbs)
set(_generated ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${_generated})
add_custom_command(OUTPUT ${_generated}/world_state_generated.h
	COMMAND ${HELIX_FLATC_COMMAND} --cpp -o ${_generated} ${_schema}
	DEPENDS ${_schema} ${HELIX_FLATC_DEPENDENCY} VERBATIM)
add_executable(helix_serialization_benchmark serialization.cpp ${_generated}/world_state_generated.h)
target_link_libraries(helix_serialization_benchmark PRIVATE helix_protocol)
target_include_directories(helix_serialization_benchmark PRIVATE ${helix_flatbuffers_SOURCE_DIR}/include ${_generated})
target_compile_features(helix_serialization_benchmark PRIVATE cxx_std_20)
if(HELIX_BUILD_FLATBUFFERS)
	add_executable(helix_wire_benchmark wire_serialization.cpp)
	add_executable(helix_wire_allocation_probe wire_serialization.cpp CountNew.cpp)
	target_compile_definitions(helix_wire_allocation_probe PRIVATE HELIX_COUNT_NEW=1)
	foreach(_target helix_wire_benchmark helix_wire_allocation_probe)
		target_link_libraries(${_target} PRIVATE helix_protocol_flatbuffers)
		target_include_directories(${_target} PRIVATE ${HELIX_SOURCE_ROOT}/src)
	endforeach()
endif()
